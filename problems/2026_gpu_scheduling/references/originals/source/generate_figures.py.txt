# Panel (a): Unconstrained Active Demand Time-Series on A10 vs A100-80G

t_days = np.arange(0, 2161) / 24.0
for m, cap, col, lbl in [('A10', 2494, '#c0392b', 'A10 Active Demand'),
                         ('A100-SXM4-80GB', 3456, '#2e86c1', 'A100-80G Active Demand')]:
    sub = job_df[job_df['gpu_model']==m]
    g_tot = (sub['gpu_request'] * sub['worker_num']).values
    s_idx = np.clip((sub['submit_time'].values // 3600).astype(int), 0, 2160)
    e_idx = np.clip((np.ceil((sub['submit_time'].values + sub['duration'].values) / 3600)).astype(int), 0, 2160)
    diff = np.zeros(2162, dtype=float)
    np.add.at(diff, s_idx, g_tot)
    np.add.at(diff, e_idx, -g_tot)
    active = np.cumsum(diff[:-1])

    ax1.plot(t_days, active_smooth, color=col, linewidth=1.6, label=lbl)
    ax1.axhline(cap, color=col, linestyle='--', linewidth=1.3, label=f'{m[:4]} Physical Cap ({cap} GPUs)')
